3.1 等待图片加载 + 懒加载滚动
async def _wait_image_loaded_playwright(self, page, image_selector, prev_src=None, timeout=10000):
    # 优先等待 src 变化
    if prev_src:
        try:
            await page.wait_for_function(
                """(sel, prev) => {
                    const el = document.querySelector(sel);
                    return el && el.src && el.src !== prev && el.complete && el.naturalWidth>10;
                }""",
                image_selector, prev_src, timeout=timeout
            )
            return
        except:
            pass
    # 兜底：只要加载完成
    await page.wait_for_function(
        """(sel) => {
            const el = document.querySelector(sel);
            return el && el.complete && el.naturalWidth>10;
        }""",
        image_selector, timeout=timeout
    )

async def _scroll_lazy_until_stable_playwright(self, page, timeout=3000):
    # 逐步滚动，直到 scrollHeight 连续稳定
    end = time.time() + (timeout/1000)
    last_h = -1; stable = 0
    while time.time() < end:
        h = await page.evaluate("() => document.documentElement.scrollHeight")
        if h == last_h:
            stable += 1
            if stable >= 3: break
        else:
            stable = 0
        last_h = h
        await page.evaluate("() => window.scrollBy(0, window.innerHeight)")
        await asyncio.sleep(0.3)

3.2 交互式裁剪（页面内鼠标框选，一次性确定）
PICK_BOX_JS = """() => new Promise(resolve => {
  const overlay = document.createElement('div');
  Object.assign(overlay.style,{position:'fixed',left:0,top:0,right:0,bottom:0,zIndex:2147483647,cursor:'crosshair'});
  document.body.appendChild(overlay);
  let sx=0, sy=0, rect=null;
  overlay.addEventListener('mousedown', e=>{
    sx=e.clientX; sy=e.clientY;
    rect=document.createElement('div');
    Object.assign(rect.style,{position:'fixed',border:'2px dashed red',background:'rgba(255,0,0,0.08)',left:sx+'px',top:sy+'px',zIndex:2147483648,width:'0px',height:'0px'});
    document.body.appendChild(rect);
    overlay.addEventListener('mousemove', onMove);
    overlay.addEventListener('mouseup', onUp);
  });
  function onMove(e){
    const x=Math.min(e.clientX,sx),y=Math.min(e.clientY,sy);
    const w=Math.abs(e.clientX-sx),h=Math.abs(e.clientY-sy);
    Object.assign(rect.style,{left:x+'px',top:y+'px',width:w+'px',height:h+'px'});
  }
  function onUp(e){
    overlay.removeEventListener('mousemove', onMove);
    overlay.removeEventListener('mouseup', onUp);
    const r=rect.getBoundingClientRect();
    const msg=document.createElement('div');
    Object.assign(msg.style,{position:'fixed',bottom:'20px',right:'20px',padding:'6px 10px',background:'#111',color:'#fff',borderRadius:'6px',zIndex:2147483648});
    msg.textContent='按 Enter 确认，Esc 取消';
    document.body.appendChild(msg);
    function cleanup(){[overlay,rect,msg].forEach(n=>n&&n.remove());}
    function onKey(ev){
      if(ev.key==='Enter'){ cleanup(); resolve({left:r.left,top:r.top,width:r.width,height:r.height}); }
      else if(ev.key==='Escape'){ cleanup(); resolve(null); }
      window.removeEventListener('keydown', onKey);
    }
    window.addEventListener('keydown', onKey);
  }
});"""

async def pick_crop_box(self, page):
    coords = await page.evaluate(PICK_BOX_JS)
    if not coords: return None
    dpr = await page.evaluate("() => window.devicePixelRatio") or 1
    l = int(round(coords['left']*dpr)); t = int(round(coords['top']*dpr))
    r = l + int(round(coords['width']*dpr)); b = t + int(round(coords['height']*dpr))
    return (l,t,r,b)

3.3 截图（元素优先；否则全页+裁剪）
async def _shoot_playwright(self, page, out_path: Path, cfg: ScreenshotConfig):
    if cfg.image_selector and await page.query_selector(cfg.image_selector):
        el = await page.query_selector(cfg.image_selector)
        await el.screenshot(path=str(out_path))
        return out_path
    # 全页
    tmp = out_path.parent / f"full_{out_path.stem}.png"
    await page.screenshot(path=str(tmp), full_page=True)
    if cfg.crop_box:
        im = Image.open(tmp); im = im.crop(cfg.crop_box); im.save(out_path)
        tmp.unlink(missing_ok=True)
    else:
        tmp.rename(out_path)
    return out_path

3.4 翻页循环（点击 next/prev / 键盘箭头 / 等 URL 或 src 变化）
async def capture_by_clicks(self, start_url: str, out_dir: Path, cfg: ScreenshotConfig,
                            max_pages: int = 2000, map_csv: Optional[Path]=None):
    out_dir.mkdir(parents=True, exist_ok=True)
    if map_csv is None: map_csv = out_dir / "mapping.csv"
    header = not map_csv.exists()
    with map_csv.open("a", encoding="utf-8", newline="") as f:
        w = csv.writer(f)
        if header: w.writerow(["seq","page_num","filename","url","timestamp"])

        page = self.playwright_page
        await page.goto(start_url, wait_until="networkidle")
        await asyncio.sleep(0.4)

        # 裁剪交互
        if cfg.interactive_crop and not cfg.crop_box:
            box = await self.pick_crop_box(page)
            if box: cfg.crop_box = box

        seq = 1
        for _ in range(max_pages):
            # 懒加载处理
            if cfg.lazy_scroll: await self._scroll_lazy_until_stable_playwright(page, int(cfg.lazy_timeout*1000))

            # 等待图片加载
            prev_src = None
            if cfg.image_selector and await page.query_selector(cfg.image_selector):
                el = await page.query_selector(cfg.image_selector)
                prev_src = await el.get_attribute("src")
                await self._wait_image_loaded_playwright(page, cfg.image_selector, prev_src=None, timeout=10000)

            # 解析页码
            url = page.url
            m = re.search(r'/n(\d+)(?:/|$)', url) or re.search(r'[?&]page=(\d+)', url)
            page_num = int(m.group(1)) if m else None
            fname = f"{page_num:04d}.png" if page_num else f"{seq:04d}.png"
            out_path = out_dir / fname

            if not out_path.exists():
                await self._shoot_playwright(page, out_path, cfg)
                w.writerow([seq, page_num or "", str(out_path), url, datetime.utcnow().isoformat()])
                self.logger.info(f"[saved] {out_path}")
            else:
                self.logger.info(f"[skip] {out_path} exists")

            # 下一页
            changed = False
            try:
                if cfg.next_selector and await page.query_selector(cfg.next_selector):
                    prev_url = page.url
                    prev_src2 = None
                    if cfg.image_selector and await page.query_selector(cfg.image_selector):
                        prev_src2 = await (await page.query_selector(cfg.image_selector)).get_attribute("src")
                    await (await page.query_selector(cfg.next_selector)).click()
                    try:
                        await self._wait_image_loaded_playwright(page, cfg.image_selector, prev_src=prev_src2, timeout=10000)
                        changed = True
                    except:
                        # 兜底：等 URL 改变或短暂等待
                        try:
                            await page.wait_for_function("old => location.href !== old", prev_url, timeout=5000)
                            changed = True
                        except:
                            await asyncio.sleep(0.5)
                elif cfg.use_arrow_keys:
                    prev_url = page.url
                    await page.keyboard.press("ArrowRight")
                    try:
                        await self._wait_image_loaded_playwright(page, cfg.image_selector, prev_src=prev_src, timeout=10000)
                        changed = True
                    except:
                        try:
                            await page.wait_for_function("old => location.href !== old", prev_url, timeout=5000)
                            changed = True
                        except:
                            await asyncio.sleep(0.5)
                else:
                    self.logger.info("没有 next_selector 或键盘翻页，结束")
                    break
            except Exception as e:
                self.logger.info(f"下一页失败/结束: {e}")
                break

            if not changed:
                self.logger.info("未检测到页变化，可能末页。结束。")
                break

            seq += 1

    return out_dir

3.5 合并→预览→询问 OCR→分别落盘到 G:\

你已有 PDF 合成功能。建议在流程入口增加一个「点击翻页工作流」包装器。

async def process_clicks_to_pdf(self, start_url: str, screenshot_cfg: ScreenshotConfig, pdf_cfg: PDFConfig):
    out_dir = Path(tempfile.mkdtemp(prefix="shots_"))
    map_csv = out_dir / "mapping.csv"

    await self.capture_by_clicks(start_url, out_dir, screenshot_cfg, map_csv=map_csv)

    # 合并前预览（打开资源管理器）
    os.startfile(str(out_dir))  # Windows only
    input("请预览图片是否正确、顺序无误。按 Enter 继续合并为PDF...")

    pdf_cfg.output_dir.mkdir(parents=True, exist_ok=True)
    pdf_path = pdf_cfg.output_dir / (pdf_cfg.title or "book.pdf")
    self.logger.info(f"合并至: {pdf_path}")
    with open(pdf_path, "wb") as f:
        f.write(img2pdf.convert([str(p) for p in sorted(out_dir.glob("*.png"))]))

    ans = input("是否进行 OCR? (Y/N): ").strip().lower()
    if ans == "y":
        ocr_out = pdf_cfg.ocr_output_dir / (pdf_cfg.title or "book_ocr.pdf")
        pdf_cfg.ocr_output_dir.mkdir(parents=True, exist_ok=True)
        cmd = f'ocrmypdf -l chi_sim --deskew "{pdf_path}" "{ocr_out}"'
        self.logger.info(f"执行: {cmd}")
        try:
            subprocess.run(cmd, shell=True, check=True)
            self.logger.info(f"OCR 输出: {ocr_out}")
        except Exception as e:
            self.logger.error(f"OCR 失败: {e}")

    return {"images_dir": str(out_dir), "pdf_path": str(pdf_path)}

@cli.command("capture-by-clicks")
@click.option('--start-url', required=True)
@click.option('--next-selector', default=None)
@click.option('--image-selector', default=None)
@click.option('--use-arrow-keys', is_flag=True, default=False)
@click.option('--interactive-crop/--no-interactive-crop', default=True)
@click.option('--merge-pdf/--no-merge-pdf', default=True)
@click.pass_context
def capture_by_clicks_cmd(ctx, **kw):
    cfg = ctx.obj['config']
    sc = ScreenshotConfig(
        image_selector=kw['image_selector'] or cfg.screenshot.element_selector,
        next_selector=kw['next_selector'] or None,
        use_arrow_keys=kw['use_arrow_keys'],
        interactive_crop=kw['interactive_crop']
    )
    pc = PDFConfig(
        output_dir=cfg.pdf.output_dir,
        # 也可以用 --pdf-name 传标题
    )
    asyncio.run(_run_clicks_flow(kw['start_url'], sc, pc))

async def _run_clicks_flow(start_url, sc, pc):
    async with WebScreenshotPDF(browser_engine="playwright") as p:
        await p.process_clicks_to_pdf(start_url, sc, pc)
